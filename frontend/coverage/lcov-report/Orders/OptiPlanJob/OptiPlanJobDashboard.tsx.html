// ─── State machine pipeline adımları ────────────────────────────────────────
const PIPELINE: OptiJobState[] = [
  "NEW", "PREPARED", "OPTI_IMPORTED", "OPTI_RUNNING", "OPTI_DONE", "XML_READY", "DELIVERED", "DONE",
];
 
const PIPELINE_LABELS: Partial<Record<OptiJobState, string>> = {
  NEW: "Yeni", PREPARED: "Hazır", OPTI_IMPORTED: "Aktarıldı",
  OPTI_RUNNING: "Çalışıyor", OPTI_DONE: "Bitti", XML_READY: "XML Hazır",
  DELIVERED: "İletildi", DONE: "Tamamlandı",
};
 
// ─── Pipeline görselleştirme ─────────────────────────────────────────────────
function JobPipeline({ state }: { state: OptiJobState }) {
  const idx = PIPELINE.indexOf(state);
  const isFailed = state === "FAILED";
  const isHold = state === "HOLD";
 
  if (isFailed || isHold) {
    const color = isFailed ? "var(--danger, #ef4444)" : "var(--warning, #f59e0b)";
    return (
      <span style={{
        display: "inline-flex", alignItems: "center", gap: 5,
        padding: "2px 10px", fontSize: 11, fontWeight: 600,
        color, background: `color-mix(in srgb, ${color} 12%, transparent)`,
        border: `1px solid color-mix(in srgb, ${color} 25%, transparent)`,
      }}>
        {isFailed ? <XCircle size={11} /> : <AlertTriangle size={11} />}
        {OPTI_STATE_LABELS[state]}
      </span>
    );
  }
 
  return (
    <div style={{ display: "flex", alignItems: "center" }}>
      {PIPELINE.map((s, i) => {
        const done   = i < idx;
        const active = i === idx;
        const dotColor = active
          ? OPTI_STATE_COLORS[s]
          : done ? "var(--success, #10b981)" : "var(--border, #333)";
        return (
          <div key={s} style={{ display: "flex", alignItems: "center" }}>
            <div
              title={PIPELINE_LABELS[s]}
              style={{
                width: 9, height: 9, borderRadius: "50%", background: dotColor, flexShrink: 0,
                boxShadow: active ? `0 0 0 3px color-mix(in srgb, ${OPTI_STATE_COLORS[s]} 25%, transparent)` : undefined,
              }}
            />
            {i < PIPELINE.length - 1 && (
              <div style={{ width: 10, height: 2, background: done ? "var(--success, #10b981)" : "var(--border, #333)" }} />
            )}
          </div>
        );
      })}
      <span style={{
        marginLeft: 8, fontSize: 10, fontWeight: 700,
        color: idx >= 0 ? OPTI_STATE_COLORS[state] : "var(--text-dim, #71717a)",
      }}>
        {PIPELINE_LABELS[state] ?? OPTI_STATE_LABELS[state]}
      </span>
    </div>
  );
}

// ─── Audit olay zaman çizelgesi ──────────────────────────────────────────────

          display: "flex", gap: 8, padding: "5px 8px",
          background: "var(--bg-main, #121212)",
          border: "1px solid var(--border, #333)", fontSize: 11,

            {ev.createdAt ? new Date(ev.createdAt).toLocaleString("tr-TR", { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit", second: "2-digit" }) : "—"}

  );
}

// ─── Ana Dashboard bileşeni ──────────────────────────────────────────────────
export function OptiPlanJobDashboard() {

  const [total, setTotal]                   = useState(0);
  const [workerStatus, setWorkerStatus]     = useState<WorkerStatus | null>(null);
  const [loading, setLoading]               = useState(true);
  const [error, setError]                   = useState<string | null>(null);
  const [expandedId, setExpandedId]         = useState<string | null>(null);
  const [auditCache, setAuditCache]         = useState<Record<string, OptiAuditEvent[]>>({});
  const [busyId, setBusyId]                 = useState<string | null>(null);
  const [stateFilter, setStateFilter]       = useState("ALL");
  const [showNewJob, setShowNewJob]         = useState(false);
  const timerRef  = useRef<ReturnType<typeof setTimeout> | null>(null);
  const mountedRef = useRef(true);
 
  const load = useCallback(async () => {
    try {
      const [list, worker] = await Promise.all([
        optiplanJobApi.listJobs({ state: stateFilter === "ALL" ? undefined : stateFilter }),
        optiplanJobApi.getWorkerStatus().catch(() => null),
      ]);
      if (!mountedRef.current) return;
      setJobs(list.jobs);
      setTotal(list.total);
      setWorkerStatus(worker);
      setError(null);
    } catch (e) {
      if (mountedRef.current) setError(e instanceof Error ? e.message : "Veri yüklenemedi");
    } finally {
      if (mountedRef.current) {
        setLoading(false);
        timerRef.current = setTimeout(() => void load(), 30_000);
      }
    }
  }, [stateFilter]);
 
  useEffect(() => {
    mountedRef.current = true;
    setLoading(true);
    void load();
    return () => {
      mountedRef.current = false;
      if (timerRef.current) clearTimeout(timerRef.current);
    };
  }, [load]);
 
  const handleExpand = useCallback(async (id: string) => {
    const next = expandedId === id ? null : id;
    setExpandedId(next);
    if (next && !auditCache[next]) {
      try {
        const events = await optiplanJobApi.getAuditEvents(next);
        setAuditCache((p) => ({ ...p, [next]: events }));
      } catch { /* sessizce geç */ }
    }
  }, [expandedId, auditCache]);
 
  const doAction = useCallback(async (id: string, fn: (id: string) => Promise<OptiJob>) => {
    setBusyId(id);
    try {
      const updated = await fn(id);
      setJobs((p) => p.map((j) => j.id === updated.id ? updated : j));
    } catch (e) {
      setError(e instanceof Error ? e.message : "İşlem başarısız");
    } finally {
      setBusyId(null);
    }
  }, []);
 
  const STATE_FILTERS = [
    { value: "ALL",          label: "Tümü" },
    { value: "NEW",          label: "Yeni" },
    { value: "OPTI_RUNNING", label: "Çalışıyor" },
    { value: "XML_READY",    label: "XML Hazır" },
    { value: "HOLD",         label: "Beklemede" },
    { value: "FAILED",       label: "Hatalı" },
    { value: "DONE",         label: "Tamamlandı" },
  ];
 
  if (showNewJob) {
    return (
      <div style={{ display: "flex", flexDirection: "column", height: "100%", background: "var(--bg-main, #121212)" }}>
        <div style={{
          padding: "8px 16px", borderBottom: "1px solid var(--border, #333)",
          background: "var(--bg-surface, #1e1e1e)", display: "flex", alignItems: "center", gap: 10,
        }}>
          <Button size="sm" variant="ghost" onClick={() => setShowNewJob(false)}>← Job Listesi</Button>
          <span style={{ fontSize: 12, color: "var(--text-muted, #94a3b8)" }}>Yeni OptiPlanning Job Oluştur</span>
        </div>
        <div style={{ flex: 1, overflow: "auto" }}>
          <OptiPlanStrictOrderEntry />
        </div>
      </div>
    );
  }
 
  return (
    <div style={{ display: "flex", flexDirection: "column", height: "100%", background: "var(--bg-main, #121212)" }}>
 
      {/* ── Başlık ── */}
      <div style={{
        padding: "14px 20px", borderBottom: "1px solid var(--border, #333)",
        background: "var(--bg-surface, #1e1e1e)",
        display: "flex", justifyContent: "space-between", alignItems: "flex-start",
        gap: 12, flexShrink: 0,
      }}>
        <div style={{ display: "flex", gap: 10, alignItems: "flex-start" }}>
          <span style={{
            width: 34, height: 34,
            background: "color-mix(in srgb, var(--primary, #38BDF8) 12%, transparent)",
            border: "1px solid color-mix(in srgb, var(--primary, #38BDF8) 20%, transparent)",
            display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0,
          }}>
            <Cpu size={16} color="var(--primary, #38BDF8)" />
          </span>
          <div>
            <h2 style={{ fontSize: 14, fontWeight: 700, color: "var(--text-main, #f3f4f6)", margin: 0 }}>
              OptiPlanning Job Kuyruğu
            </h2>
            <p style={{ fontSize: 11, color: "var(--text-muted, #94a3b8)", margin: "2px 0 10px" }}>
              {total} kayıt · 30 saniyede bir otomatik güncellenir
            </p>
            <WorkerStatusBar status={workerStatus} />
          </div>
        </div>
        <div style={{ display: "flex", gap: 8, flexShrink: 0 }}>
          <Button
            size="sm" variant="ghost"
            icon={<RefreshCw size={12} style={{ animation: loading ? "spin 1s linear infinite" : "none" }} />}
            onClick={() => { setLoading(true); void load(); }}
            disabled={loading}
          >
            Yenile
          </Button>
          <Button size="sm" variant="primary" icon={<Plus size={13} />} onClick={() => setShowNewJob(true)}>
            Yeni Job
          </Button>
        </div>
      </div>
 
      {/* ── Durum filtreleri ── */}
      <div style={{
        padding: "8px 20px", borderBottom: "1px solid var(--border, #333)",
        background: "var(--bg-surface, #1e1e1e)",
        display: "flex", gap: 6, alignItems: "center", flexWrap: "wrap", flexShrink: 0,
      }}>
        <span style={{ fontSize: 10, fontWeight: 700, color: "var(--text-dim, #71717a)",
          textTransform: "uppercase", letterSpacing: "0.06em" }}>DURUM:</span>
        {STATE_FILTERS.map((f) => {
          const active = stateFilter === f.value;
          return (
            <button key={f.value} type="button" onClick={() => setStateFilter(f.value)} style={{
              padding: "3px 10px", fontSize: 11, cursor: "pointer",
              border: `1px solid ${active ? "var(--primary, #38BDF8)" : "var(--border, #333)"}`,
              background: active ? "color-mix(in srgb, var(--primary, #38BDF8) 12%, transparent)" : "transparent",
              color: active ? "var(--primary, #38BDF8)" : "var(--text-muted, #94a3b8)",
              fontWeight: active ? 700 : 400,
            }}>
              {f.label}
            </button>
          );
        })}
      </div>
 
      {/* ── İçerik ── */}
      <div style={{ flex: 1, overflow: "auto" }}>
        {error && (
          <div style={{
            margin: "12px 20px", padding: "10px 14px",
            background: "color-mix(in srgb, var(--danger, #ef4444) 8%, transparent)",
            border: "1px solid color-mix(in srgb, var(--danger, #ef4444) 25%, transparent)",
            fontSize: 12, color: "var(--danger, #ef4444)", display: "flex", gap: 8,
          }}>
            <AlertTriangle size={14} style={{ flexShrink: 0, marginTop: 1 }} />{error}
          </div>
        )}
 
        {loading && jobs.length === 0 ? (
          <div style={{ display: "flex", alignItems: "center", justifyContent: "center",
            height: 200, gap: 10, color: "var(--text-muted, #94a3b8)", fontSize: 13 }}>
            <RefreshCw size={16} style={{ animation: "spin 1s linear infinite" }} />
            Yükleniyor…
          </div>
        ) : jobs.length === 0 ? (
          <div style={{ display: "flex", flexDirection: "column", alignItems: "center",
            justifyContent: "center", padding: "60px 20px", textAlign: "center" }}>
            <Cpu size={36} color="var(--text-dim, #71717a)" />
            <p style={{ marginTop: 12, fontSize: 13, color: "var(--text-muted, #94a3b8)" }}>
              {stateFilter === "ALL"
                ? "Henüz optimizasyon job'u yok"
                : `"${STATE_FILTERS.find((f) => f.value === stateFilter)?.label}" durumunda job yok`}
            </p>
            {stateFilter === "ALL" && (
              <div style={{ marginTop: 16 }}>
                <Button size="sm" variant="primary" icon={<Plus size={13} />} onClick={() => setShowNewJob(true)}>
                  İlk Job'u Oluştur
                </Button>
              </div>
            )}
          </div>
        ) : (
          <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 12 }}>
            <thead>
              <tr style={{ background: "var(--bg-surface, #1e1e1e)" }}>
                <th style={TH} />
                <th style={TH}>Job ID</th>
                <th style={TH}>Sipariş</th>
                <th style={TH}>İlerleme Durumu</th>
                <th style={TH}>Hata</th>
                <th style={TH}>Oluşturulma</th>
                <th style={{ ...TH, textAlign: "right" }}>Aksiyonlar</th>
              </tr>
            </thead>
            <tbody>
              {jobs.map((job) => (
                <JobRow
                  key={job.id}
                  job={job}
                  expanded={expandedId === job.id}
                  onToggle={() => void handleExpand(job.id)}
                  onRetry={(id) => void doAction(id, optiplanJobApi.retryJob)}
                  onApprove={(id) => void doAction(id, optiplanJobApi.approveJob)}
                  onCancel={(id) => void doAction(id, optiplanJobApi.cancelJob)}
                  onDownload={(id) => window.open(optiplanJobApi.getXmlDownloadUrl(id), "_blank")}
                  auditEvents={auditCache[job.id] ?? []}
                  busyId={busyId}
                />
              ))}
            </tbody>
          </table>
        )}
      </div>
 
      {/* ── Script uyarısı ── */}
      {workerStatus && !workerStatus.scriptExists && (
        <div style={{
          padding: "7px 20px", borderTop: "1px solid color-mix(in srgb, var(--warning, #f59e0b) 30%, transparent)",
          background: "color-mix(in srgb, var(--warning, #f59e0b) 5%, transparent)",
          fontSize: 11, color: "var(--warning, #f59e0b)",
          display: "flex", gap: 6, alignItems: "center", flexShrink: 0,
        }}>
          <AlertTriangle size={12} />
          OptiPlanning script bulunamadı:
          <code style={{ fontFamily: "var(--font-mono)" }}>{workerStatus.scriptPath}</code>
        </div>
      )}
 
      <style>{`@keyframes spin { from { transform: rotate(0deg); } to { transform: rotate(360deg); } }`}</style>
    </div>
  );
}
