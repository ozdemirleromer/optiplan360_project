import { nanoid } from "nanoid";
import type {
  OrderEntryOrder,
  OrderEntryItem,
  OrderEntrySelectedCell,
} from "./types";
 
export function createEmptyOrderItem(): OrderEntryItem {
  return {
    id: nanoid(),
    stok_kod: "",
    stok_ad: "",
    material: "",
    length: "",
    width: "",
    quantity: "",
    grain: 0,
    info: "",
    edge_L1: false,
    edge_L2: false,
    edge_W1: false,
    edge_W2: false,
    edge_thickness: "",
    hole_1: "",
    hole_2: "",
  };
}
 
export function createEmptyOrder(orderId: string): OrderEntryOrder {
  return {
    id: orderId,
    customerId: "",
    crmAccountId: "",
    customerName: "",
    materialDefault: "",
    plateSize: "",
    generalNote: "",
    deliveryDate: null,
    deliveryAddress: null,
    paymentMethod: null,
    status: "draft",
    createdBy: "",
    lockedByUserId: null,
    lockedByName: null,
    lockedAt: null,
    currentUserCanEdit: true,
    items: [createEmptyOrderItem()],
  };
}
 
export function cloneOrderEntryOrder(order: OrderEntryOrder): OrderEntryOrder {
  return {
    ...order,
    items: order.items.map((item) => ({ ...item })),
  };
}
 
export function duplicateOrderItem(item: OrderEntryItem): OrderEntryItem {
  return { ...item, id: nanoid() };
}
 
export function insertRowAfter(
  items: OrderEntryItem[],
  afterId: string | null
): OrderEntryItem[] {
  const newItem = createEmptyOrderItem();
  if (afterId === null || items.length === 0) {
    return [...items, newItem];
  }
  const index = items.findIndex((i) => i.id === afterId);
  if (index === -1) return [...items, newItem];
  const next = [...items];
  next.splice(index + 1, 0, newItem);
  return next;
}
 
export function removeGridRow(
  items: OrderEntryItem[],
  rowId: string
): OrderEntryItem[] {
  if (items.length <= 1) return items;
  return items.filter((i) => i.id !== rowId);
}
 
export function moveItem(
  items: OrderEntryItem[],
  rowId: string,
  dir: "up" | "down"
): OrderEntryItem[] {
  const index = items.findIndex((i) => i.id === rowId);
  if (index === -1) return items;
  if (dir === "up" && index === 0) return items;
  if (dir === "down" && index === items.length - 1) return items;
 
  const next = [...items];
  const swapIndex = dir === "up" ? index - 1 : index + 1;
  [next[index], next[swapIndex]] = [next[swapIndex], next[index]];
  return next;
}
 
export function hasOrderChanged(
  prev: OrderEntryOrder,
  next: OrderEntryOrder
): boolean {
  return JSON.stringify(prev) !== JSON.stringify(next);
}
 
export function isReadonlyOrder(order: OrderEntryOrder): boolean {
  return !order.currentUserCanEdit;
}
 
export function formatLockSummary(order: OrderEntryOrder): string {
  if (!order.lockedByName) return "";
  return `Bu sipariş ${order.lockedByName} tarafından düzenleniyor`;
}
 
export function getDefaultSelectedCell(
  order: OrderEntryOrder | null
): OrderEntrySelectedCell | null {
  if (!order || order.items.length === 0) return null;
  return { rowId: order.items[0].id, column: "material" };
}
 
export function focusGridCell(cell: OrderEntrySelectedCell | null): void {
  if (!cell) return;
  const el = document.querySelector<HTMLElement>(
    `[data-order-cell="${cell.rowId}:${cell.column}"]`
  );
  el?.focus();
}
